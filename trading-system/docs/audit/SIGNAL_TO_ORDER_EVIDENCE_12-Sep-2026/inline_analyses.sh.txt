import pandas as pd, numpy as np, itertools
ORDER = [("volume_surge",15),("vwap_position",10),("atr_filter",10),("rsi_range",10),("price_action",15),
         ("sector_strength",10),("time_of_day",5),("spread_check",5),("circuit_check",10),("signal_age",10)]
def score(d):
    tot=0.0
    for k,w in ORDER: tot += float(d[k])*w
    return min(100,int(round((tot/100)*100.0)))
# Enumerate the live discrete steps (dead ones at their constants) and find the minimum price_action that reaches 60
rows=[]
for vw,tod,sp,cc,age in itertools.product([0,1],[0.5,0.8,1.0],[0.0,0.5,1.0],[0,1],[0.0,0.5,1.0]):
    base=dict(volume_surge=0,vwap_position=vw,atr_filter=0,rsi_range=0.5,sector_strength=0.5,time_of_day=tod,spread_check=sp,circuit_check=cc,signal_age=age)
    need=None
    for p in np.arange(0,1.00001,0.0001):
        d=dict(base,price_action=float(round(p,4)))
        if score(d)>=60: need=round(p,4); break
    mx=score(dict(base,price_action=1.0))
    rows.append((vw,tod,sp,cc,age,mx,need))
t=pd.DataFrame(rows,columns=["vwap","time_of_day","spread","circuit","age","max_score","min_price_action_to_pass"])
