#!/usr/bin/env python3
"""READ-ONLY: did any DECISION outcome change on the twin with Batch 1 running?
(1) THE DECISION MIX since the restart against the same clock window on the previous
    trading day: signals by status, screener verdicts by status, trades and orders
    created. Live traffic differs day to day, so this shows the same decision vocabulary
    in comparable proportions -- it is NOT a counterfactual proof.
(2) AGREEMENT: every evidence row against the decision the pipeline itself recorded for
    that signal_id -- P2 vs signals.status, P3 vs screener_results.status, P1 vs the
    signal's status. The observer must RECORD the decision, never alter it.
Writes nothing (DB read-only, no bytecode).   usage: python -u - [since HH:MM:SS]"""
import collections
import glob
import json

ROOT = "/home/ubuntu/systems/trading-system"
SINCE = sys.argv[1] if len(sys.argv) > 1 else "09:16:44"
host = socket.gethostname()
print("host: %s | time: %s | window: %s -> now" % (host, time.strftime("%F %T %Z"), SINCE))
if host != "trading-system-sandbox":
    print("ABORT: not the Testing VM")
    sys.exit(9)
os.chdir(ROOT)

today, now = time.strftime("%Y-%m-%d"), time.strftime("%H:%M:%S")
prev = con.execute("SELECT MAX(substr(received_at,1,10)) FROM signals WHERE substr(received_at,1,10) < ?",
                   (today,)).fetchone()[0]


def pct(n, t):
    return "%5.1f%%" % (100.0 * n / t) if t else "   - "


print("\n== (1) THE DECISION MIX, same clock window: %s vs %s ==" % (prev, today))

    a, b = "%sT%s" % (day, SINCE), "%sT%s" % (day, now)
    tot = con.execute("SELECT COUNT(*) FROM signals WHERE received_at >= ? AND received_at < ?", (a, b)).fetchone()[0]
    print("\n  %s  %s -> %s   signals received: %d" % (day, SINCE, now, tot))
    for st, n in con.execute("SELECT status, COUNT(*) FROM signals WHERE received_at >= ? AND received_at < ? "
                             "GROUP BY 1 ORDER BY 2 DESC LIMIT 14", (a, b)):
        print("    signals     %-34s %6d %s" % (st, n, pct(n, tot)))
    stot = con.execute("SELECT COUNT(*) FROM screener_results WHERE ts >= ? AND ts < ?", (a, b)).fetchone()[0]
    for st, n in con.execute("SELECT status, COUNT(*) FROM screener_results WHERE ts >= ? AND ts < ? GROUP BY 1 "
                             "ORDER BY 2 DESC LIMIT 8", (a, b)):
        print("    screener    %-34s %6d %s" % (st, n, pct(n, stot)))
    for st, n in con.execute("SELECT status, COUNT(*) FROM trades WHERE created_at >= ? AND created_at < ? GROUP BY 1",
                             (a, b)):
        print("    trades      %-34s %6d" % (st, n))
    for leg, st, n in con.execute("SELECT leg, status, COUNT(*) FROM orders WHERE placed_at >= ? AND placed_at < ? "
                                  "GROUP BY 1, 2", (a, b)):
        print("    orders      %-34s %6d" % (leg + "/" + st, n))

print("\n== (2) AGREEMENT: each evidence row vs the decision the pipeline recorded ==")
rows = []
for f in sorted(glob.glob("data_store/evidence/signal_evidence_*_%s.jsonl" % today)):
    if "failure" in os.path.basename(f):
        continue
    for line in open(f, encoding="utf-8", errors="replace"):
        try:

        res[(cp, "no signal_id")] += 1
        continue
    s = con.execute("SELECT status FROM signals WHERE signal_id = ?", (sid,)).fetchone()

        if s is None:
            res[(cp, "no signals row")] += 1
        elif st == s[0]:
            res[(cp, "AGREE")] += 1
        else:
            res[(cp, "DIFFER")] += 1
            examples.append((cp, sid, st, s[0]))
    elif cp == "P3_SCREEN":
        sr = set(x[0] for x in con.execute("SELECT status FROM screener_results WHERE signal_id = ?", (sid,)))
        if not sr:
            res[(cp, "no screener_results row")] += 1
        elif st in sr:
            res[(cp, "AGREE")] += 1
        else:
            res[(cp, "DIFFER")] += 1
            examples.append((cp, sid, st, sorted(sr)))
    elif cp == "P1_ACCEPT":
        p1_db[s[0] if s else "no signals row"] += 1
        res[(cp, "checked")] += 1
for k in sorted(res, key=str):
    print("  %-40s %6d" % ("%s %s" % k, res[k]))
if p1_db:
    print("  P1 rows -> the signal's status in the DB: %s" % dict(p1_db))
for e in examples[:10]:
    print("  DIFFER: %s signal %s row status=%r db=%r" % e)
print("\nDONE")

#!/usr/bin/env python3
"""The per-record cost of the evidence recorder's write on THIS machine's volume (v2 --
the OUTLIER profile, 09:35 file sec 5.2): the same open(append) + write + flush + fsync
the recorder performs per record, with a realistic JSON line, measured while the trading
service runs (its load is the background). Also the same without fsync, to isolate it.
Reports the mean AND the tail -- p99, max, the five worst, and how many exceeded 5 ms /
10 ms / 20 ms -- because one 16 ms stall on the decision path matters more than a good
average. NOTE: a disk-level PROXY: it measures this volume under the service's concurrent
load, NOT the in-process cost on the trading thread (lock contention included) -- so it
informs S11-R19 and does not close it. Writes ONLY its own temp file, in a directory it
creates under /home/ubuntu/preserved (the same filesystem as data_store), and removes
both at the end.   usage: python -u - [records] [line_bytes] [spacing_ms]"""
import json
import os
import socket
import statistics
import subprocess
import sys
import time

N = int(sys.argv[1]) if len(sys.argv) > 1 else 600
SIZE = int(sys.argv[2]) if len(sys.argv) > 2 else 2500
GAP = (float(sys.argv[3]) if len(sys.argv) > 3 else 20.0) / 1000.0
host = socket.gethostname()
print("host: %s | time: %s | records %d x %d bytes, spacing %.0f ms" % (host, time.strftime("%F %T %Z"), N, SIZE, GAP * 1000))
if host != "trading-system-sandbox":
    print("ABORT: not the Testing VM")
    sys.exit(9)
d = "/home/ubuntu/preserved/fsync_bench_%s" % time.strftime("%Y%m%d_%H%M%S")
os.mkdir(d)
path = os.path.join(d, "bench.jsonl")
print("same device as data_store: %s" % (os.stat(d).st_dev == os.stat("/home/ubuntu/systems/trading-system/data_store").st_dev))
line = json.dumps({"pad": "x" * max(0, SIZE - 12)}) + "\n"


def sh(c):
    return subprocess.run(c, shell=True, capture_output=True, text=True).stdout.strip()


def run(do_fsync):
    out = []
    for _ in range(N):
        t0 = time.perf_counter()
        with open(path, "a", encoding="utf-8") as fh:
            fh.write(line)
            fh.flush()
            if do_fsync:
                os.fsync(fh.fileno())
        out.append((time.perf_counter() - t0) * 1000.0)
        time.sleep(GAP)
    return out


def summ(label, ts):
    s = sorted(ts)
    q = lambda p: s[min(len(s) - 1, int(round(p * (len(s) - 1))))]  # noqa: E731
    print("  %-28s mean %7.3f ms | p50 %7.3f | p95 %7.3f | p99 %7.3f | max %7.3f  (n=%d)" % (
        label, statistics.mean(s), q(0.50), q(0.95), q(0.99), s[-1], len(s)))
    print("  %-28s >5 ms: %d | >10 ms: %d | >20 ms: %d | five worst: %s" % (
        "", sum(1 for x in s if x > 5), sum(1 for x in s if x > 10), sum(1 for x in s if x > 20),
        ", ".join("%.1f" % x for x in s[-5:][::-1])))
