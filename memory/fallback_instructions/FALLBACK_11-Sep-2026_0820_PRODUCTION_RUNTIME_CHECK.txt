text = raw_b.decode("utf-8", errors="replace").replace("\r\n", "\n")


def find(pattern):
    found = re.findall(pattern, text, re.M)
    return found[-1].strip() if found else None


if "== TARGET HOST ==" not in text:
    sys.exit("NO RECORD: the script never started on the host (no '== TARGET HOST ==' line) -- a "
             "transport failure, not a check result. The raw output stays at %s." % RAW)
host = find(r"^  hostname  : (.+)$")
prod = find(r"^  production: (\S+)")
if host is not None and (host != "trading-system" or prod != "True"):
    sys.exit("REFUSED (P0): the raw output is from host %r (production=%s) -- not production "
             "evidence; no record is written" % (host, prod))
for p in (MD, SIDE):
    if os.path.exists(p):
        sys.exit("REFUSED: %s exists -- a record is never overwritten; a correction is a NEW "
                 "record that cites it" % p)
open(SIDE, "wb").write(raw_b)
os.chmod(SIDE, stat.S_IREAD)
sha_raw = hashlib.sha256(raw_b).hexdigest()
print("preserved the raw output FIRST: %s (%d bytes, sha256 %s)" % (SIDE, len(raw_b), sha_raw))

try:
    complete = "\nDONE" in text
    NR = "(not reached -- the check did not complete)"

    tblock = between("== TRIAGE", "== VERDICT")
    vblock = between("== VERDICT", "\nDONE")
    sha_scr = hashlib.sha256(open(SCRIPT, "rb").read()).hexdigest()

            "| status | " + status + " |\n"
            "| target host — the first thing the script printed | `" + host_s + "` (production's hostname "

            "| run by | " + RUN_BY + " |\n"
            "| raw output | `" + NAME + ".raw.txt` beside this file — written FIRST, before this record; " +
