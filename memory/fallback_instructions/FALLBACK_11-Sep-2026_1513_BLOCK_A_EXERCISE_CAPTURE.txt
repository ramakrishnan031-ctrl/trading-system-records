#!/usr/bin/env python3
"""READ-ONLY 15:03 Block-A exercise capture, AFTER THE FACT (v3).
CLOSE-SIX sec 5.2, extended by the 11-Sep 05:30 file sec 4 and the 11-Sep 06:35
file sec 4: a host guard, the five fill classes, pass identity by broker_order_id.

Did PASS_1 (15:03) / PASS_2 (15:06) find any MIS position on production today?
  none -> NOT EXECUTED is never the word here: the result is NOT EXERCISED. Never
          PASS, never CLEAN.
  any  -> per symbol: the scan line, the exact pass submitted, the market_protection
          value, the broker_order_id, broker status and history, filled / pending /
          cancelled quantity, average price, EVERY order on that symbol today (so a
          manual close is never counted as the system's), the final broker position,
          and each order's class: FULL / PARTIAL + residual / NONE + resting /
          CANCELLED / REJECTED.
The pass is identified by broker_order_id, NEVER by the tag: both passes truncate
to "mis_autosq_pass_" (S11-R7). A flat PASS_2 proves NOTHING about PASS_1 (10-Sep
IDEA: PASS_2 found the book flat only because Rama had closed it by hand).
The host is printed FIRST; on any host but production nothing here is a
production result.
Writes NOTHING: no files, no orders, no cancels; the DB is opened read-only. Prints
no account id. ASCII output, flushed per line.   usage: python -u - [--no-broker]
The broker's order book is DAILY: run this on the same trading day, or the
broker-side half of the evidence is gone for good."""
import functools
import json
import os
import re
import sqlite3
import subprocess
import sys
from datetime import date

print = functools.partial(print, flush=True)  # noqa: A001

ROOT = "/home/ubuntu/systems/trading-system"
EXPECT_HOST = "trading-system"            # production; measured 11-Sep 06:33 via ssh trading-vm
SANDBOX_HOST = "trading-system-sandbox"   # the Testing VM
TODAY = date.today().isoformat()
LOG = os.path.join(ROOT, "logs", "system_%s.log" % TODAY)
DB = os.path.join(ROOT, "data_store", "trading_system.db")
TOKEN = os.path.join(ROOT, "data_store", "session", "zerodha_token.json")
NO_BROKER = "--no-broker" in sys.argv
SENT_KEYS = ("pass", "symbol", "broker_order_id", "qty", "side", "order_type",
             "market_protection_percent", "pass_1_order_id", "residual_detail")


def a(s):
    return str(s).encode("ascii", "replace").decode("ascii")


def sh(c):
    return subprocess.run(c, shell=True, capture_output=True, text=True).stdout.strip()


def field(line, name):
    m = re.search(r'"%s"\s*:\s*("([^"]*)"|[^,}\s]+)' % re.escape(name), line)
    if not m:
        return None
    return m.group(2) if m.group(2) is not None else m.group(1)


def classify(o):
    qty, fq = o.get("quantity") or 0, o.get("filled_quantity") or 0
    pend, canc, st = o.get("pending_quantity") or 0, o.get("cancelled_quantity") or 0, o.get("status")
    if st == "REJECTED":
        return "REJECTED"
    if st == "COMPLETE" and fq == qty:
        return "FULL"
    if 0 < fq < qty:
        return "PARTIAL + residual %d (%s)" % (
            qty - fq, "resting, pending=%d" % pend if pend else "cancelled=%d" % canc)
    if fq == 0 and pend:
        return "NONE + resting (pending=%d, status %s)" % (pend, st)
    if st == "CANCELLED":
        return "CANCELLED"
    return "UNCLASSIFIED (status %s, filled %s of %s)" % (st, fq, qty)


host = sh("hostname")
is_prod = host == EXPECT_HOST
print("== TARGET HOST ==")
print("  hostname  : %s" % a(host))
print("  production: %s   (production's hostname is %r)" % (is_prod, EXPECT_HOST))
if not is_prod:
    print("  >>> NOT PRODUCTION%s -- NOTHING BELOW IS A PRODUCTION RESULT <<<"
          % (" -- this is the Testing VM" if host == SANDBOX_HOST else ""))
print("  time      : %s" % a(sh("date '+%F %T %Z'")))
print("log: %s (%s)" % (LOG, "present" if os.path.exists(LOG) else "MISSING"))

scan, submitted, states = [], [], []
if os.path.exists(LOG):
    with open(LOG, encoding="utf-8", errors="replace") as fh:
        for line in fh:
            if "MIS_AUTO_SQUAREOFF_SCAN" in line:
                scan.append(line.strip())
            elif "MIS_AUTO_SQUAREOFF_EXIT_SUBMITTED" in line:
                submitted.append(line.strip())
            elif "mis_autosquareoff " in line and re.search(
                    r"EXIT_|RESTORE_|PASS_|BROKER_STATE|DEADLINE|RESIDUAL|CANCEL_", line):
                states.append(line.strip())

print("\n== MIS_AUTO_SQUAREOFF_SCAN -- one per pass ==")
for l in scan:
    print("  pass=%s mis_candidates=%s | %s" % (field(l, "pass"), field(l, "mis_candidates"),
                                               a(l)[:300]))
print("\n== MIS_AUTO_SQUAREOFF_EXIT_SUBMITTED -- what was SENT ==")
sent, pass_by_id = {}, {}
for l in submitted:
    d = dict((k, field(l, k)) for k in SENT_KEYS)
    sent.setdefault(d["symbol"], []).append(d)
    pass_by_id[str(d["broker_order_id"])] = d["pass"]
    print("  " + " ".join("%s=%s" % (k, a(v)) for k, v in d.items()))
    print("    raw: %s" % a(l)[:500])
print("\n== mis_autosquareoff state lines (first 80) ==")
for l in states[:80]:
    print("  " + a(l)[:420])

print("\n== local trades created today with an MIS entry (read-only; NOT the authority) ==")
try:
    wal = os.path.exists(DB + "-wal")
    con = sqlite3.connect("file:%s?mode=ro%s" % (DB, "" if wal else "&immutable=1"),
                          uri=True, timeout=5)
    rows = con.execute(
        "SELECT t.symbol, t.status, t.qty_filled, t.created_at FROM trades t "
        "WHERE t.created_at LIKE ? AND (SELECT o.product FROM orders o WHERE o.trade_id = "
        "t.trade_id AND o.leg = 'ENTRY' LIMIT 1) = 'MIS' ORDER BY t.created_at",
        (TODAY + "%",)).fetchall()
    print("  %d trade(s)" % len(rows))
    for r in rows:
        print("    %-14s %-12s qty_filled=%s created=%s" % tuple(a(x) for x in r))
    con.close()
except Exception as e:  # noqa: BLE001
    print("  DB read FAILED -- %s: %s" % (type(e).__name__, a(e)))

print("\n== broker truth (READ-ONLY: profile, positions, orders, order history) ==")
summary = []
if NO_BROKER:
    print("  skipped (--no-broker)")
else:
    try:
        from kiteconnect import KiteConnect
        tok = json.load(open(TOKEN))
        kc = KiteConnect(api_key=tok["api_key"])
        kc.set_access_token(tok["access_token"])
        kc.profile()
        print("  profile read OK (account id deliberately not printed)")
        pos = kc.positions() or {}
        day_mis = [p for p in (pos.get("day") or []) if p.get("product") == "MIS"]
        net_mis = dict((p.get("tradingsymbol"), p) for p in (pos.get("net") or [])
                       if p.get("product") == "MIS")
        orders = kc.orders() or []
        sq = [o for o in orders if str(o.get("tag") or "").startswith("mis_autosq")
              or str(o.get("order_id")) in pass_by_id]
        syms = sorted(set(p.get("tradingsymbol") for p in day_mis)
                      | set(o.get("tradingsymbol") for o in sq) | set(s for s in sent if s))
        print("  MIS day-book rows: %d | squareoff orders (by id or tag): %d | symbols examined: %s"
              % (len(day_mis), len(sq), " ".join(a(s) for s in syms) or "none"))
        for s in syms:
            n = net_mis.get(s) or {}
            q = n.get("quantity") or 0
            final = "FLAT" if q == 0 else "RESIDUAL %s" % q
            print("\n  -- %s: FINAL net MIS qty=%s (buy=%s sell=%s avg=%s pnl=%s) => %s" % (
                a(s), q, n.get("buy_quantity"), n.get("sell_quantity"), n.get("average_price"),
                n.get("pnl"), final))
            for d in sent.get(s, []):
                print("     SENT (log): pass=%s broker_order_id=%s qty=%s side=%s type=%s "
                      "market_protection=%s%%" % (d["pass"], d["broker_order_id"], d["qty"],
                                                  d["side"], d["order_type"],
                                                  d["market_protection_percent"]))
            classes = []
            for o in sorted([o for o in orders if o.get("tradingsymbol") == s],
                            key=lambda o: str(o.get("order_timestamp"))):
                oid, tag = str(o.get("order_id")), str(o.get("tag") or "")
                if oid in pass_by_id:
                    who = "SYSTEM squareoff %s (by broker_order_id)" % pass_by_id[oid]
                elif tag.startswith("mis_autosq"):
                    who = "squareoff-tagged, pass UNKNOWN (tag truncated -- S11-R7)"
                else:
                    who = "NOT the squareoff (tag %r)" % a(tag)
                mine = oid in pass_by_id or tag.startswith("mis_autosq")
                cls = classify(o) if mine else "-"
                if mine:
                    classes.append("%s:%s" % (pass_by_id.get(oid, "PASS_?"), cls))
                print("     %s id=%s %s %s %s qty=%s filled=%s pending=%s cancelled=%s avg=%s "
                      "status=%s mp=%s [%s] class=%s" % (
                          o.get("order_timestamp"), oid, o.get("transaction_type"),
                          o.get("product"), o.get("order_type"), o.get("quantity"),
                          o.get("filled_quantity"), o.get("pending_quantity"),
                          o.get("cancelled_quantity"), o.get("average_price"), o.get("status"),
                          o.get("market_protection"), who, cls))
                if o.get("status_message"):
                    print("        msg=%s" % a(o.get("status_message"))[:300])
                if mine:
                    for h in kc.order_history(oid) or []:
                        print("        %s %-16s filled=%s pending=%s avg=%s %s" % (
                            h.get("order_timestamp"), h.get("status"), h.get("filled_quantity"),
                            h.get("pending_quantity"), h.get("average_price"),
                            a(h.get("status_message"))[:160]))
            summary.append("%s final=%s squareoff orders=[%s]" % (a(s), final,
                                                                  "; ".join(classes) or "none"))
    except Exception as e:  # noqa: BLE001
        print("  broker read FAILED -- %s: %s  (report it; infer NOTHING from the log alone)"
              % (type(e).__name__, a(e)))

print("\n== verdict inputs ==")
cand = []
for l in scan:
    m = re.search(r'"mis_candidates"\s*:\s*(\d+)', l) or re.search(r"mis_candidates=(\d+)", l)
    if m:
        cand.append(int(m.group(1)))
print("  host=%s production=%s | scan lines=%d mis_candidates parsed=%s exits submitted=%d"
      % (a(host), is_prod, len(scan), cand, len(submitted)))
for line in summary:
    print("  per symbol: " + line)
if not is_prod:
    print("  => NOT A PRODUCTION RESULT (host %s)." % a(host))
elif not scan:
    print("  => NO SCAN LINE: the passes did not run or did not log. Report exactly that -- it "
          "is neither 'flat' nor a pass.")
elif len(cand) != len(scan):
    print("  => SCAN LINES PRESENT BUT mis_candidates NOT PARSED -- read them by eye; do NOT "
          "call the book flat on a failed parse.")
elif all(c == 0 for c in cand) and not submitted:
    print("  => NOT EXERCISED -- every pass logged mis_candidates=0 and nothing was submitted. "
          "Confirm against the broker MIS rows above. NEVER PASS, NEVER CLEAN.")
else:
    print("  => EXERCISED -- per symbol, what was SENT against what HAPPENED:")
    print("     EXERCISED - CLEAN  only if the system's protected MARKET filled FULL and the")
    print("                        final position is FLAT because of that order;")
    print("     EXERCISED - FAILED for REJECTED, PARTIAL + residual, NONE + resting, CANCELLED,")
    print("                        or a final position that is not flat;")
    print("     NOT EXERCISED      for a symbol the squareoff never sent an order for.")
    print("     A flat PASS_2 proves NOTHING about PASS_1's order (10-Sep IDEA).")
print("\nDONE")
