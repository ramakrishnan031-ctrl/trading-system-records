#!/usr/bin/env python3
"""READ-ONLY: the MIS book FROM THE BROKER (10:45 file sec 1.1) -- "local state is not the fact that
decides whether an exercise happens". Prints the broker's net MIS positions (quantity, average,
last price, P&L), EVERY order today on each MIS symbol in order_timestamp order (so a natural SL/TGT
fill or a manual close is visible as what it is), every non-zero net position of any product, and
the local DB's live trades BESIDE it so the lag shows. Runs on either VM and NAMES the machine; a
result from one machine is never the other's. Writes nothing (no bytecode, DB mode=ro, no files);
prints no account id.   usage: python -u -"""
import json
import os
import socket
import sqlite3
import sys
import time

sys.dont_write_bytecode = True
HOSTS = {"trading-system": "PRODUCTION (trading VM, LFL836)", "trading-system-sandbox": "TESTING VM (twin, VBB097)"}
host = socket.gethostname()
print("host: %s = %s | time: %s" % (host, HOSTS.get(host, "UNKNOWN"), time.strftime("%F %T %Z")))
if host not in HOSTS:
    print("ABORT: not one of the two known VMs")
    sys.exit(9)
ROOT = "/home/ubuntu/systems/trading-system"
os.chdir(ROOT)
sys.path.insert(0, ROOT)
from kiteconnect import KiteConnect  # noqa: E402

tok = json.load(open("data_store/session/zerodha_token.json"))
kc = KiteConnect(api_key=tok["api_key"])
kc.set_access_token(tok["access_token"])
pos = kc.positions() or {}
net = pos.get("net") or []
mis = [p for p in net if p.get("product") == "MIS"]
open_mis = [p for p in mis if p.get("quantity")]
print("\n== MIS positions, broker net book: %d row(s), %d OPEN (quantity != 0) ==" % (len(mis), len(open_mis)))
for p in mis:
    print("  %-12s qty=%-5s avg=%-10s ltp=%-10s pnl=%-9s buy_qty=%s sell_qty=%s" % (
        p.get("tradingsymbol"), p.get("quantity"), p.get("average_price"), p.get("last_price"),
        round(p.get("pnl") or 0, 2), p.get("buy_quantity"), p.get("sell_quantity")))
orders = kc.orders() or []
syms = sorted(set(p.get("tradingsymbol") for p in mis))
print("\n== every order today on those symbols (broker), by order_timestamp: ==")
for o in sorted((o for o in orders if o.get("tradingsymbol") in syms), key=lambda o: str(o.get("order_timestamp"))):
    print("  %s %-12s %-4s %-3s %-6s %-16s qty=%s filled=%s trig=%s px=%s avg=%s id=%s tag=%s" % (
        o.get("order_timestamp"), o.get("tradingsymbol"), o.get("transaction_type"), o.get("product"),
        o.get("order_type"), o.get("status"), o.get("quantity"), o.get("filled_quantity"), o.get("trigger_price"),
        o.get("price"), o.get("average_price"), o.get("order_id"), o.get("tag")))
nz = [p for p in net if p.get("quantity")]
print("\n== every non-zero net position, any product: %d ==" % len(nz))
for p in nz:
    print("  %-12s %-4s qty=%s" % (p.get("tradingsymbol"), p.get("product"), p.get("quantity")))
db = "data_store/trading_system.db"
con = sqlite3.connect("file:%s?mode=ro%s" % (db, "" if os.path.exists(db + "-wal") else "&immutable=1"), uri=True)
live = con.execute("SELECT t.symbol, t.status, o.product, t.created_at FROM trades t LEFT JOIN orders o "
                   "ON o.trade_id = t.trade_id AND o.leg = 'ENTRY' WHERE t.status IN "
                   "('PENDING_FILL','OPEN','PARTIAL','EXITING') ORDER BY t.created_at").fetchall()
print("\n== the local DB's live trades (the lag, NOT the fact): %d ==" % len(live))
for r in live:
    print("  %-12s %-8s %-4s created %s" % (r[0], r[1], r[2], str(r[3])[11:19]))
bro = sorted(p.get("tradingsymbol") for p in open_mis)
loc = sorted(r[0] for r in live if r[2] == "MIS")
print("\nBROKER MIS OPEN: %s | local MIS live: %s | agree: %s" % (bro, loc, bro == loc))
